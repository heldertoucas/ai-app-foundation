### WP-01 Proteger `/api/open-terminal`

- **Modelo:** S
- **Depende:** WP-00
- **Validação:** NÃO PILOTADO
- **Ficheiros:** `lib/security/local-request.ts` (novo), `app/api/open-terminal/route.ts`, `tests/local-request.test.ts` (novo)

**Porquê:** remover só o CORS não basta. Um POST `text/plain` não faz preflight e `request.json()` aceita-o.

**1. Criar `lib/security/local-request.ts`:**

```ts
const ALLOWED_ORIGINS = new Set(['http://localhost:4300', 'http://127.0.0.1:4300']);
const ALLOWED_HOSTS = new Set(['localhost:4300', '127.0.0.1:4300']);

/**
 * Accepts only same-machine browser requests sent as JSON.
 * Requiring JSON forces a CORS preflight, which this app never approves.
 */
export function isTrustedLocalRequest(request: Request): boolean {
  const origin = request.headers.get('origin');
  const host = request.headers.get('host');
  const contentType = request.headers.get('content-type')?.split(';')[0]?.trim().toLowerCase();

  return Boolean(
    origin &&
      host &&
      ALLOWED_ORIGINS.has(origin) &&
      ALLOWED_HOSTS.has(host) &&
      contentType === 'application/json',
  );
}
```

**2. Substituir `app/api/open-terminal/route.ts` por:**

```ts
import { NextResponse } from 'next/server';
import { execFile } from 'child_process';
import os from 'os';
import { isTrustedLocalRequest } from '@/lib/security/local-request';

const ALLOWED_COMMANDS = new Set([
  'pnpm dev',
  'pnpm start',
  'pnpm foundation:check',
  'open public/html-kit/templates/du-avaliacao-dashboard.html',
  'open public/html-kit/templates/du-dashboard.html',
  'open public/html-kit/templates/starter-cockpit.html',
  'open public/html-kit/templates/template-app-shell.html',
  'open public/html-kit/templates/du-html-app-template.html',
]);

export async function POST(request: Request) {
  if (!isTrustedLocalRequest(request)) {
    return NextResponse.json({ success: false, error: 'Origem não autorizada.' }, { status: 403 });
  }

  if (os.platform() !== 'darwin') {
    return NextResponse.json(
      { success: false, error: 'Abertura direta de terminal desativada por conformidade em ambientes não-macOS.' },
      { status: 403 },
    );
  }

  try {
    const body = await request.json();
    const command = String(body.command ?? '').trim();

    if (!ALLOWED_COMMANDS.has(command)) {
      return NextResponse.json(
        { success: false, error: 'Comando não autorizado na whitelist de execução.' },
        { status: 400 },
      );
    }

    const workDir = process.cwd();
    const script = `tell application "Terminal" to do script "cd '${workDir}' && ${command}" activate`;

    execFile('osascript', ['-e', script], (err) => {
      if (err) console.error('[open-terminal] Erro ao disparar Terminal.app:', err);
    });

    return NextResponse.json({ success: true, command });
  } catch {
    return NextResponse.json({ success: false, error: 'Pedido inválido.' }, { status: 400 });
  }
}
```

**3. Criar `tests/local-request.test.ts`:**

```ts
import { describe, expect, it } from 'vitest';
import { isTrustedLocalRequest } from '@/lib/security/local-request';

const request = (headers: Record<string, string>) =>
  new Request('http://localhost:4300/api/open-terminal', { method: 'POST', headers });
const valid = { origin: 'http://localhost:4300', host: 'localhost:4300', 'content-type': 'application/json' };

describe('isTrustedLocalRequest', () => {
  it('accepts a same-machine JSON request', () => expect(isTrustedLocalRequest(request(valid))).toBe(true));
  it('accepts JSON with a charset', () =>
    expect(isTrustedLocalRequest(request({ ...valid, 'content-type': 'application/json; charset=utf-8' }))).toBe(true));
  it('rejects an external origin', () =>
    expect(isTrustedLocalRequest(request({ ...valid, origin: 'https://evil.example' }))).toBe(false));
  it('rejects a null origin', () => expect(isTrustedLocalRequest(request({ ...valid, origin: 'null' }))).toBe(false));
  it('rejects a missing origin', () => {
    const { origin: _origin, ...rest } = valid;
    expect(isTrustedLocalRequest(request(rest))).toBe(false);
  });
  it('rejects DNS rebinding through Host', () =>
    expect(isTrustedLocalRequest(request({ ...valid, host: 'evil.example:4300' }))).toBe(false));
  it('rejects text/plain requests that skip preflight', () =>
    expect(isTrustedLocalRequest(request({ ...valid, 'content-type': 'text/plain' }))).toBe(false));
});
```
