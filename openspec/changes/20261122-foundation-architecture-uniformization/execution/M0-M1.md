### WP-01 Proteger `/api/open-terminal`

- **Modelo:** S
- **Depende:** WP-03
- **Validação:** NÃO PILOTADO

**Porquê:** remover só o CORS não basta. Um POST `text/plain` não faz preflight e `request.json()` aceita-o.

**1. Criar `lib/security/local-request.ts`:**

```ts
const ALLOWED_ORIGINS = new Set(['http://localhost:4300', 'http://127.0.0.1:4300']);
const ALLOWED_HOSTS = new Set(['localhost:4300', '127.0.0.1:4300']);

export function isAllowedHost(host: string | null): boolean {
  return Boolean(host && ALLOWED_HOSTS.has(host));
}

/** Validates browser origin and Host. It depends on the server being bound to loopback (WP-07). */
export function isTrustedLocalOrigin(headers: Headers): boolean {
  const origin = headers.get('origin');
  return Boolean(origin && ALLOWED_ORIGINS.has(origin) && isAllowedHost(headers.get('host')));
}

/** JSON forces a CORS preflight, which this app never approves. */
export function isTrustedLocalRequest(request: Request): boolean {
  const contentType = request.headers.get('content-type')?.split(';')[0]?.trim().toLowerCase();
  return isTrustedLocalOrigin(request.headers) && contentType === 'application/json';
}
```

**2. Substituir `app/api/open-terminal/route.ts` por:**

```ts
import { NextResponse } from 'next/server';
import { execFile } from 'node:child_process';
import os from 'node:os';
import { promisify } from 'node:util';
import { isTrustedLocalRequest } from '@/lib/security/local-request';

const execFileAsync = promisify(execFile);

const ALLOWED_COMMANDS = new Set([
  'pnpm dev',
  'pnpm start',
  'pnpm foundation:check',
  'open public/html-kit/templates/du-avaliacao-dashboard.html',
  'open public/html-kit/templates/du-dashboard.html',
  'open public/html-kit/templates/starter-cockpit.html',
  'open public/html-kit/templates/template-app-shell.html',
  'open public/html-kit/templates/du-html-app-template.html',
]);

const OPEN_TERMINAL_SCRIPT = [
  'on run argv',
  'set commandText to "cd " & quoted form of (item 1 of argv) & " && " & (item 2 of argv)',
  'tell application "Terminal"',
  'activate',
  'do script commandText',
  'end tell',
  'end run',
];

export async function POST(request: Request) {
  if (!isTrustedLocalRequest(request)) {
    return NextResponse.json({ success: false, error: 'Origem não autorizada.' }, { status: 403 });
  }

  if (os.platform() !== 'darwin') {
    return NextResponse.json(
      { success: false, error: 'Abertura direta de terminal desativada por conformidade em ambientes não-macOS.' },
      { status: 403 },
    );
  }

  let command: string;
  try {
    const body = await request.json();
    command = String(body.command ?? '').trim();
  } catch {
    return NextResponse.json({ success: false, error: 'Pedido inválido.' }, { status: 400 });
  }

  if (!ALLOWED_COMMANDS.has(command)) {
    return NextResponse.json(
      { success: false, error: 'Comando não autorizado na whitelist de execução.' },
      { status: 400 },
    );
  }

  try {
    const args = OPEN_TERMINAL_SCRIPT.flatMap((line) => ['-e', line]);
    await execFileAsync('osascript', [...args, process.cwd(), command], { timeout: 5000 });
    return NextResponse.json({ success: true, command });
  } catch (err) {
    console.error('[open-terminal] Erro ao disparar Terminal.app:', err);
    return NextResponse.json({ success: false, error: 'Não foi possível abrir o Terminal.' }, { status: 500 });
  }
}
```

import { NextResponse, type NextRequest } from 'next/server';
import { isAllowedHost } from '@/lib/security/local-request';

function isEncodedBlueprintPath(pathname: string): boolean {
  try {
    const decoded = decodeURIComponent(pathname);
    return decoded !== pathname && decoded.toLowerCase().startsWith('/blueprints/');
  } catch {
    return true;
  }
}

export function proxy(request: NextRequest) {
  if (!isAllowedHost(request.headers.get('host'))) {
    return new NextResponse('Host não autorizado.', { status: 403 });
  }
  if (isEncodedBlueprintPath(request.nextUrl.pathname)) {
    return new NextResponse('Não encontrado.', { status: 404 });
  }
  return NextResponse.next();
}
